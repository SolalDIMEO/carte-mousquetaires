// ============================================================================ // Fonction Azure : lit en direct l'onglet "Suivi" et "Suivi prospect" du fichier // Excel stocké sur SharePoint, et renvoie un JSON prêt à consommer par la carte // (index.html). Authentification "application" via Microsoft Graph : personne // n'a besoin de se connecter — la fonction lit avec ses propres droits. // // Variables d'environnement à renseigner dans Azure (Configuration > Application // settings) — voir README.md pour savoir où trouver chaque valeur : // TENANT_ID identifiant de votre tenant Microsoft 365 // CLIENT_ID identifiant de l'application enregistrée dans Azure AD // CLIENT_SECRET secret généré pour cette application // SITE_ID identifiant du site SharePoint contenant le fichier // ITEM_ID identifiant du fichier Excel (le .xlsm) sur ce site // ALLOWED_ORIGIN (optionnel) domaine autorisé en CORS, ex: https://votrecompte.github.io // à défaut, "*" (tout le monde) est utilisé. // ============================================================================ let cache = { data: null, ts: 0 }; const CACHE_MS = 5 * 60 * 1000; // 5 minutes : évite de solliciter Graph à chaque visite module.exports = async function (context, req) { const origin = process.env.ALLOWED_ORIGIN || "*"; const cors = { "Access-Control-Allow-Origin": origin, "Content-Type": "application/json; charset=utf-8" }; try { if (cache.data && (Date.now() - cache.ts) < CACHE_MS) { context.res = { status: 200, headers: cors, body: cache.data }; return; } const token = await getAppToken(); const suiviRows = await readWorksheet(token, 'Suivi'); const prospectRows = await readWorksheet(token, 'Suivi prospect'); const payload = { suivi: rowsToObjects(suiviRows, SUIVI_HEADERS), prospect: rowsToObjects(prospectRows, PROSPECT_HEADERS), generatedAt: new Date().toISOString() }; cache = { data: payload, ts: Date.now() }; context.res = { status: 200, headers: cors, body: payload }; } catch (err) { context.log.error(err); context.res = { status: 500, headers: cors, body: { error: "Lecture du fichier impossible", detail: String(err && err.message || err) } }; } }; // --------------------------------------------------------------------------- // Authentification "application" (client credentials) — aucun utilisateur // n'est impliqué, la fonction agit avec ses propres droits (admin consent requis). // --------------------------------------------------------------------------- async function getAppToken() { const url = `https://login.microsoftonline.com/${process.env.TENANT_ID}/oauth2/v2.0/token`; const body = new URLSearchParams({ client_id: process.env.CLIENT_ID, client_secret: process.env.CLIENT_SECRET, scope: "https://graph.microsoft.com/.default", grant_type: "client_credentials" }); const res = await fetch(url, { method: "POST", body }); if (!res.ok) throw new Error("Échec d'authentification Azure AD (" + res.status + ")"); const json = await res.json(); return json.access_token; } // --------------------------------------------------------------------------- // Lecture d'un onglet via l'API Workbook de Graph (session non persistante, // adaptée à la lecture ponctuelle d'un classeur macro .xlsm). // --------------------------------------------------------------------------- async function readWorksheet(token, sheetName) { const base = `https://graph.microsoft.com/v1.0/sites/${process.env.SITE_ID}/drive/items/${process.env.ITEM_ID}/workbook`; const headers = { Authorization: `Bearer ${token}`, "Content-Type": "application/json" }; const sessionRes = await fetch(`${base}/createSession`, { method: "POST", headers, body: JSON.stringify({ persistChanges: false }) }); if (!sessionRes.ok) throw new Error("Impossible d'ouvrir une session sur le classeur (" + sessionRes.status + ")"); const { id: sessionId } = await sessionRes.json(); headers["workbook-session-id"] = sessionId; const rangeRes = await fetch( `${base}/worksheets('${encodeURIComponent(sheetName)}')/usedRange(valuesOnly=true)`, { headers } ); if (!rangeRes.ok) throw new Error(`Lecture de l'onglet "${sheetName}" impossible (` + rangeRes.status + ")"); const rangeJson = await rangeRes.json(); fetch(`${base}/closeSession`, { method: "POST", headers }).catch(() => {}); return rangeJson.values || []; } // --------------------------------------------------------------------------- // Correspondance colonnes <-> mêmes positions que dans le fichier Excel d'origine // (onglets "Suivi" et "Suivi prospect"). À ajuster si les colonnes changent. // --------------------------------------------------------------------------- const SUIVI_HEADERS = [ "projet","coords","puissance","type","commentaires","deadlinePA","paEnvoyee","nbPA","paValidee", "plannerCree","deadlineR2","offreEnvoyee","nbR2","offreValidee","projetFini", "commercial1","commercial2","commentaire2", null, "statut" ]; const PROSPECT_HEADERS = ["projet","statut","commercial1","commercial2","commentaire"]; function rowsToObjects(rows, headers) { if (!rows.length) return []; const dataRows = rows.slice(1); // la première ligne est l'en-tête Excel return dataRows .filter(r => r[0] !== "" && r[0] !== null && r[0] !== undefined) .map(r => { const obj = {}; headers.forEach((key, i) => { if (!key) return; const v = r[i]; if (key === "coords") { // Colonne "Coordonnées" au format "lat, lng" if (v) { const parts = String(v).split(",").map(s => parseFloat(s.trim())); obj.coords = (parts.length === 2 && !isNaN(parts[0]) && !isNaN(parts[1])) ? parts : null; } else { obj.coords = null; } return; } obj[key] = (v === null || v === undefined) ? "" : String(v).trim(); }); return obj; }); }
